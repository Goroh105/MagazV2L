-- Создание базы данных (если еще не создана)
CREATE DATABASE IF NOT EXISTS computers_db;

-- Подключаемся к базе данных
\c computers_db;

-- Создание таблицы Product
CREATE TABLE Product (
    model VARCHAR(50) PRIMARY KEY,
    maker VARCHAR(50) NOT NULL,
    type VARCHAR(20) NOT NULL CHECK (type IN ('PC', 'Laptop', 'Printer'))
);

-- Создание таблицы PC
CREATE TABLE PC (
    id SERIAL PRIMARY KEY,
    model VARCHAR(50) REFERENCES Product(model),
    speed INTEGER NOT NULL,
    ram INTEGER NOT NULL,
    hd INTEGER NOT NULL,
    cd VARCHAR(20),
    price DECIMAL(10, 2) NOT NULL,
    count INTEGER NOT NULL
);

-- Создание таблицы Laptop
CREATE TABLE Laptop (
    id SERIAL PRIMARY KEY,
    model VARCHAR(50) REFERENCES Product(model),
    speed INTEGER NOT NULL,
    ram INTEGER NOT NULL,
    hd INTEGER NOT NULL,
    screen DECIMAL(4, 1) NOT NULL,
    price DECIMAL(10, 2) NOT NULL,
    count INTEGER NOT NULL
);

-- Создание таблицы Printer
CREATE TABLE Printer (
    id SERIAL PRIMARY KEY,
    model VARCHAR(50) REFERENCES Product(model),
    color BOOLEAN NOT NULL,  -- true для цветного, false для черно-белого
    type VARCHAR(20) NOT NULL, --например, 'Laser', 'Inkjet'
    price DECIMAL(10, 2) NOT NULL,
    count INTEGER NOT NULL
);

-- Добавление комментариев к таблицам и полям (опционально, но рекомендуется)
COMMENT ON TABLE Product IS 'Таблица информации о товарах';

COMMENT ON TABLE Laptop IS 'Таблица информации о ноутбуках';
COMMENT ON COLUMN Laptop.id IS 'Суррогатный ключ';
COMMENT ON COLUMN Laptop.model IS 'Модель (внешний ключ, ссылка на Product)';
COMMENT ON COLUMN Laptop.speed IS 'Быстродействие процессора';

COMMENT ON COLUMN Laptop.price IS 'Цена';

COMMENT ON TABLE Printer IS 'Таблица информации о печатающих устройствах';
